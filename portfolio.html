<script>


let state = JSON.parse(localStorage.getItem('portfolio_v5')) || {

  cash:10000,

  holdings:[

    {ticker:'QQQ', price:508.23, shares:10, target:30},

    {ticker:'META', price:585.4, shares:5, target:20},

    {ticker:'SCHD', price:89.12, shares:60, target:50},

  ]

};


function save(){

  localStorage.setItem('portfolio_v5', JSON.stringify(state));

}


function render(){


  let total = state.cash;


  state.holdings.forEach(h=>{

    total += (h.price||0)*(h.shares||0);

  });


  document.getElementById('totalAssets').innerText =

    '$' + Math.round(total).toLocaleString();


  const tbody = document.getElementById('tbody');

  tbody.innerHTML = '';


  state.holdings.forEach((h,i)=>{


    const value = (h.price||0)*(h.shares||0);

    const targetValue = total * (h.target||0)/100;

    const diff = targetValue - value;


    const currentWeight = total>0 ? (value/total*100) : 0;


    let weightClass = '';

    if(currentWeight >= 25) weightClass='high';

    else if(currentWeight <=10) weightClass='low';


    let sharesToTrade = 0;

    if(h.price>0){

      sharesToTrade = diff>0

        ? Math.floor(diff/h.price)

        : Math.ceil(diff/h.price);

    }


    const tr = document.createElement('tr');


    tr.innerHTML = `

      <td><input class="ticker" value="${h.ticker}" onchange="edit(${i},'ticker',this.value)"></td>

      <td><input type="number" value="${h.price}" onchange="edit(${i},'price',this.value)"></td>

      <td><input type="number" value="${h.shares}" onchange="edit(${i},'shares',this.value)"></td>

      <td class="weight ${weightClass}">${currentWeight.toFixed(1)}%</td>

      <td class="cell-value">$${Math.round(value).toLocaleString()}</td>

      <td><input type="number" value="${h.target}" onchange="edit(${i},'target',this.value)"></td>

      <td class="${diff>=0?'high':'low'}">

        ${diff>=0?'+':''}$${Math.round(diff).toLocaleString()}

      </td>

      <td>

        ${sharesToTrade===0?'-':

          (sharesToTrade>0

            ? `<span style="color:green">BUY ${sharesToTrade}</span>`

            : `<span style="color:red">SELL ${Math.abs(sharesToTrade)}</span>`

          )

        }

      </td>

      <td><button onclick="del(${i})">✕</button></td>

    `;


    tbody.appendChild(tr);

  });


  save();

}


function edit(i,key,val){

  if(key==='ticker'){

    state.holdings[i][key]=val.toUpperCase();

  }else{

    state.holdings[i][key]=Number(val);

  }

  render();

}


function add(){

  state.holdings.push({ticker:'',price:0,shares:0,target:0});

  render();

}


function del(i){

  state.holdings.splice(i,1);

  render();

}


render();
